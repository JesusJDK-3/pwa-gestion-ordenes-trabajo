ESTRUCTURA DETALLADA DE TABLAS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

┌─────────────────────────────────────────────────────────────────────────┐
│ TABLE: op_ot_acompanante                                                │
├─────────────────────────────────────────────────────────────────────────┤
│                                                                          │
│ Propósito: Registrar trabajadores que acompañan al capataz en una OT   │
│ Máximo: 10 acompañantes por OT (validado por trigger)                  │
│ Rol: Fijo "AYUDANTE" (no seleccionable)                                │
│                                                                          │
│ CAMPOS:                                                                  │
│  • id_ot_acompanante    BIGSERIAL       PRIMARY KEY                    │
│  • id_ot                BIGINT          NOT NULL, FK → op_orden_trabajo│
│  • id_trabajador        BIGINT          FK → rrhh_trabajador [OPTIONAL]│
│  • dni                  VARCHAR(8)      [OPTIONAL pero si no FK]       │
│  • nombres              VARCHAR(100)    NOT NULL                        │
│  • apellidos            VARCHAR(100)    NOT NULL                        │
│  • cargo                VARCHAR(100)    DEFAULT 'AYUDANTE'             │
│  • rol                  VARCHAR(30)     NOT NULL DEFAULT 'AYUDANTE'    │
│  • orden_en_lista       INT             NOT NULL DEFAULT 0             │
│  • activo               BOOLEAN         NOT NULL DEFAULT TRUE          │
│  • created_at           TIMESTAMP       NOT NULL DEFAULT NOW()         │
│  • updated_at           TIMESTAMP       NOT NULL DEFAULT NOW()         │
│                                                                          │
│ RESTRICCIONES:                                                           │
│  • UNIQUE (id_ot, id_trabajador) WHERE id_trabajador IS NOT NULL      │
│    └─ Un trabajador no puede aparecer dos veces en la misma OT        │
│  • CHECK (id_trabajador IS NOT NULL OR dni IS NOT NULL)              │
│    └─ O bien referencia a trabajador existente O bien DNI             │
│  • PRIMARY KEY: id_ot_acompanante (BIGSERIAL autoincremento)          │
│                                                                          │
│ INDICES:                                                                 │
│  • idx_ot_acompanante_ot ..................... (id_ot)                │
│  • idx_ot_acompanante_trabajador ............ (id_trabajador)        │
│  • idx_ot_acompanante_orden ................. (id_ot, orden_en_lista) │
│                                                                          │
│ TRIGGERS:                                                                │
│  • trg_validar_limite_acompanantes .......... BEFORE INSERT          │
│    └─ Valida: no más de 10 acompañantes activos por OT               │
│                                                                          │
└─────────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────────┐
│ TABLE: op_ot_purgado_hidrante                                           │
├─────────────────────────────────────────────────────────────────────────┤
│                                                                          │
│ Propósito: Datos técnicos del purgado de hidrantes                      │
│ Relación: Un purgado por formulario técnico                             │
│ Validación: Campos obligatorios cuando formulario está COMPLETADO       │
│                                                                          │
│ CAMPOS:                                                                  │
│  • id_purgado               BIGSERIAL       PRIMARY KEY                │
│  • id_ot_formulario         BIGINT          NOT NULL, FK (UNIQUE)      │
│  • id_ot                    BIGINT          NOT NULL, FK (desnormal.)  │
│  • marca_hidrante           VARCHAR(100)    [OPCIONAL]                │
│  • numero_bocamazas         INT             [OPCIONAL]                │
│  • presion_psi_hidrante     DECIMAL(10,2)   [OPCIONAL]                │
│  • tiempo_inicio_purgado    TIMESTAMP       [OPCIONAL]                │
│  • tiempo_fin_purgado       TIMESTAMP       [OPCIONAL]                │
│  • medicion_cloro_ppm       DECIMAL(10,2)   [OPCIONAL]                │
│  • observaciones            TEXT            [OPCIONAL]                │
│  • created_at               TIMESTAMP       NOT NULL DEFAULT NOW()    │
│  • updated_at               TIMESTAMP       NOT NULL DEFAULT NOW()    │
│                                                                          │
│ RESTRICCIONES:                                                           │
│  • UNIQUE (id_ot_formulario)                                           │
│    └─ Solo un purgado por formulario técnico                          │
│  • PRIMARY KEY: id_purgado (BIGSERIAL autoincremento)                 │
│                                                                          │
│ INDICES:                                                                 │
│  • idx_purgado_formulario ................... (id_ot_formulario, UNIQ)│
│  • idx_purgado_ot ........................... (id_ot)                │
│  • idx_purgado_fecha ........................ (created_at)            │
│                                                                          │
│ TRIGGERS:                                                                │
│  • trg_validar_purgado_obligatorios ........ BEFORE INSERT/UPDATE     │
│    └─ Si formulario COMPLETADO: todos los campos deben estar llenos   │
│    └─ Campos requeridos:                                               │
│        • marca_hidrante                                                 │
│        • numero_bocamazas                                               │
│        • presion_psi_hidrante                                          │
│        • tiempo_inicio_purgado                                          │
│        • tiempo_fin_purgado                                            │
│        • medicion_cloro_ppm                                            │
│                                                                          │
└─────────────────────────────────────────────────────────────────────────┘


FUNCIONES TRIGGER
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

┌─────────────────────────────────────────────────────────────────────────┐
│ FUNCTION: fn_validar_limite_acompanantes()                              │
├─────────────────────────────────────────────────────────────────────────┤
│                                                                          │
│ PROPÓSITO:                                                               │
│   Validar que no haya más de 10 acompañantes activos por OT            │
│                                                                          │
│ DISPARADOR:                                                              │
│   trigger: trg_validar_limite_acompanantes                             │
│   evento: BEFORE INSERT en op_ot_acompanante                           │
│   para cada fila                                                         │
│                                                                          │
│ LOGICA:                                                                  │
│   1. Contar acompañantes activos en la OT actual (NEW.id_ot)           │
│   2. Si count >= 10:                                                    │
│      - Lanzar excepción: "No se pueden agregar más de 10 acompañantes" │
│   3. Si count < 10:                                                     │
│      - Permitir el INSERT                                               │
│                                                                          │
│ RESULTADO:                                                               │
│   - INSERT es bloqueado si hay 10 o más acompañantes activos           │
│   - Se puede editar/eliminar acompañantes sin restricción               │
│                                                                          │
└─────────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────────┐
│ FUNCTION: fn_validar_purgado_obligatorios()                             │
├─────────────────────────────────────────────────────────────────────────┤
│                                                                          │
│ PROPÓSITO:                                                               │
│   Validar que los campos técnicos estén llenos si el formulario está   │
│   en estado "COMPLETADO"                                                │
│                                                                          │
│ DISPARADOR:                                                              │
│   trigger: trg_validar_purgado_obligatorios                            │
│   evento: BEFORE INSERT o UPDATE en op_ot_purgado_hidrante             │
│   para cada fila                                                         │
│                                                                          │
│ LOGICA:                                                                  │
│   1. Obtener estado_formulario de op_ot_formulario                     │
│   2. Si estado = 'COMPLETADO':                                          │
│      - Validar que TODOS estos campos NO sean NULL:                     │
│        • marca_hidrante                                                 │
│        • numero_bocamazas                                               │
│        • presion_psi_hidrante                                          │
│        • tiempo_inicio_purgado                                          │
│        • tiempo_fin_purgado                                            │
│        • medicion_cloro_ppm                                            │
│      - Si alguno es NULL: lanzar excepción                             │
│   3. Actualizar updated_at = NOW()                                     │
│   4. Retornar fila modificada                                           │
│                                                                          │
│ RESULTADO:                                                               │
│   - INSERT/UPDATE bloqueado si formula COMPLETADO pero faltan campos   │
│   - Permite insertar datos incompletos si formulario no está COMPLETADO │
│   - updated_at se actualiza automáticamente en cada operación          │
│                                                                          │
└─────────────────────────────────────────────────────────────────────────┘

INDICES Y OPTIMIZACIONES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Tabla: op_ot_acompanante
───────────────────────

  1. idx_ot_acompanante_ot
     ┌─ Optimiza: SELECT * FROM op_ot_acompanante WHERE id_ot = ?
     ├─ Uso: Obtener todos los acompañantes de una OT
     └─ Cardinality: Alta (muchos acompañantes por OT)

  2. idx_ot_acompanante_trabajador
     ┌─ Optimiza: SELECT * FROM op_ot_acompanante WHERE id_trabajador = ?
     ├─ Uso: Encontrar en qué OTs participa un trabajador
     └─ Cardinality: Alta

  3. idx_ot_acompanante_orden
     ┌─ Optimiza: SELECT * FROM op_ot_acompanante 
     │            WHERE id_ot = ? ORDER BY orden_en_lista
     ├─ Uso: Obtener acompañantes ordenados por posición en lista
     └─ Cardinality: Media (compuesto)


Tabla: op_ot_purgado_hidrante
──────────────────────────────

  1. idx_purgado_formulario (UNIQUE)
     ┌─ Optimiza: SELECT * FROM op_ot_purgado_hidrante 
     │            WHERE id_ot_formulario = ?
     ├─ Uso: Obtener purgado de un formulario específico (PK alternativa)
     └─ Cardinality: Baja (UNO por formulario)

  2. idx_purgado_ot
     ┌─ Optimiza: SELECT * FROM op_ot_purgado_hidrante WHERE id_ot = ?
     ├─ Uso: Obtener todos los purgados de una OT
     └─ Cardinality: Media

  3. idx_purgado_fecha
     ┌─ Optimiza: SELECT * FROM op_ot_purgado_hidrante 
     │            WHERE created_at BETWEEN ? AND ?
     ├─ Uso: Reportes por rango de fechas
     └─ Cardinality: Alta (muchos registros por fecha)

CICLO DE VIDA DE DATOS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Acompañante:
  Estado: NUEVO
    │
    ├─► INSERT en op_ot_acompanante
    │   ├─ id_ot: requerido (FK)
    │   ├─ id_trabajador: opcional (FK)
    │   ├─ dni: si no hay id_trabajador
    │   └─ activo: true (por defecto)
    │
    ├─► VALIDACION en trigger
    │   └─ Contar activos en OT < 10 ?
    │       ├─ SÍ: Permitir INSERT
    │       └─ NO: Lanzar error
    │
    ├─► ACTIVO
    │   └─ UPDATE: Cambiar nombres, apellidos, orden_en_lista
    │
    └─► INACTIVO (soft delete)
        └─ UPDATE activo = FALSE (no eliminar realmente)


Purgado:
  Estado: PARCIAL (Formulario en creación)
    │
    ├─► INSERT en op_ot_purgado_hidrante
    │   ├─ id_ot_formulario: requerido (FK)
    │   ├─ id_ot: requerido (FK desnormalizado)
    │   └─ Campos técnicos: opcionales
    │
    ├─► SIN VALIDACION (trigger permite parcial)
    │
    ├─► COMPLETAR DATOS
    │   ├─ UPDATE marca_hidrante = ?
    │   ├─ UPDATE numero_bocamazas = ?
    │   ├─ UPDATE presion_psi_hidrante = ?
    │   ├─ UPDATE tiempo_inicio_purgado = ?
    │   ├─ UPDATE tiempo_fin_purgado = ?
    │   └─ UPDATE medicion_cloro_ppm = ?
    │
    ├─► MARCAR FORMULARIO COMO COMPLETADO
    │   └─ UPDATE op_ot_formulario SET estado_formulario = 'COMPLETADO'
    │
    ├─► PROXIMA UPDATE EN PURGADO
    │   └─ VALIDACION en trigger
    │       └─ Todos los campos requeridos están llenos ?
    │           ├─ SÍ: Permitir UPDATE
    │           └─ NO: Lanzar error
    │
    └─► COMPLETO Y VALIDADO
        └─ Datos pueden ser consultados y reportados
